# ── Vehicles ──────────────────────────────────────────────
# Each authenticated car may read/write ONLY its own VIN subtree.
# %u = MQTT username. With `use_identity_as_username true`, %u is the
# client-cert CN, which MUST equal the bare VIN (CN=TESTVIN123, NOT
# car_TESTVIN123) so it matches the vehicle/<VIN>/# scheme.
pattern readwrite vehicle/%u/#

# ── Backend service ───────────────────────────────────────
# Fleet-wide access — this is the documented "skeleton key" credential.
# Granted explicitly by identity, since the pattern above would otherwise
# confine it to vehicle/backend-service/#.
user backend-service
topic readwrite vehicle/#
